<script>

    // List of words for the game

    const WORDS = [

        "apple", "orange", "banana", "grape", "lemon", 

        // ... Add more words as needed

    ];


    let currentRow = 0;

    let guessWords = [];

    let usedWords = [];


    // Populate the letter select elements

    function populateLetters() {

        const letters = 'abcdefghijklmnopqrstuvwxyz';

        for (let i = 0; i < 26; i++) {

            const letter = letters[i];

            document.getElementById(`guess-letter${i + 1}`).innerHTML += `<option value="${letter}">${letter}</option>`;

        }

    }


    // Initialize the game

    function initializeGame() {

        populateLetters();

        document.getElementById('submit-guess').addEventListener('click', submitGuess);

    }


    // Submit the guess

    function submitGuess() {

        const guess = [];

        for (let i = 1; i <= 5; i++) {

            guess.push(document.getElementById(`guess-letter${i}`).value);

        }

        

        // Check if the guess is a valid word

        if (!WORDS.includes(guess.join(''))) {

            alert("Invalid word. Please enter a real five-letter English word.");

            return;

        }


        if(usedWords.includes(guess.join(''))){

            alert("You already guessed that word. Please try another one.");

            return;

        }


        usedWords.push(guess.join(''));


        guessWords[currentRow] = guess.join('');

        currentRow++;


        if(currentRow === 6){

            checkWord();

            return;

        }


        resetGuessContainer();

    }


    // Check the word against the secret word

    function checkWord() {

        const secretWord = WORDS[Math.floor(Math.random() * WORDS.length)];

        let feedback = "";


        for(let i = 0; i < 5; i++){

            if(guessWords[currentRow - 1][i] === secretWord[i]){

                feedback += "🟩";

            } else if(secretWord.includes(guessWords[currentRow - 1][i])){

                feedback += "🟨";

            } else {

                feedback += "⬛";

            }

        }


        guessWords[currentRow - 1].forEach((letter, index) => {

            const letterElement = document.getElementById(`letter${currentRow}-${index + 1}`);

            letterElement.innerHTML = `<span style="background-color: #${feedback[index]}">${letter}</span>`;

        });


        if(feedback === "🟩🟩🟩🟩🟩"){

            alert("Congratulations! You guessed the secret word!");

            resetGame();

        } else if(currentRow === 6){

            alert(`Sorry, the secret word was "${secretWord}".`);

            resetGame();

        }

    }


    // Reset the guess container

    function resetGuessContainer() {

        const guessRow = document.getElementById(`guess-${currentRow}`);

        if(!guessRow) return;

        

        for(let i = 1; i <= 5; i++){

            const letterElement = document.getElementById(`letter${currentRow}-${i}`);

            letterElement.innerHTML = `<span style="background-color: #f0f0f0">&nbsp;</span>`;

        }

    }


 // Reset the game

 function resetGame() {

    currentRow = 0;

    guessWords = [];

    usedWords = [];

    populateLetters();

 }


    const guessContainer = document.querySelector('.guess-container');

    while(guessContainer.firstChild) {

        guessContainer.removeChild(guessContainer.firstChild);

    }

}


// Initialize the game

initializeGame();
